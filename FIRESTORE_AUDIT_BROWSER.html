// FIRESTORE READ-ONLY AUDIT SCRIPT
// Paste this into browser console (F12 → Console tab) while app is open

(async function auditFirestore() {
  console.log('\n' + '='.repeat(70));
  console.log('FIRESTORE DATABASE AUDIT - READ-ONLY');
  console.log('='.repeat(70) + '\n');

  // Import Firebase functions
  const { getFirestore, collection, getDocs, doc, getDoc } = await import('https://www.gstatic.com/firebasejs/10.0.0/firestore.js');
  const { initializeApp } = await import('https://www.gstatic.com/firebasejs/10.0.0/app.js');

  // Get the existing app and database
  const db = getFirestore();

  console.log('📋 STEP 1: List all top-level collections');
  console.log('─'.repeat(70));
  
  try {
    const collections = await db.listCollections();
    console.log(`✅ Found ${collections.length} top-level collection(s):\n`);
    
    const collectionNames = [];
    collections.forEach((col, i) => {
      console.log(`  ${i + 1}. ${col.id}`);
      collectionNames.push(col.id);
    });
    
    console.log('\n' + '─'.repeat(70));
    console.log('🔍 STEP 2: Check for /users collection');
    console.log('─'.repeat(70) + '\n');
    
    if (collectionNames.includes('users')) {
      console.log('✅ /users COLLECTION EXISTS\n');
      
      const usersSnap = await getDocs(collection(db, 'users'));
      console.log(`   Documents in /users: ${usersSnap.docs.length}\n`);
      
      if (usersSnap.docs.length > 0) {
        console.log('   Found documents:');
        usersSnap.docs.forEach((doc, i) => {
          const data = doc.data();
          console.log(`\n   ${i + 1}. ${doc.id}`);
          console.log(`      name: ${data.name || 'N/A'}`);
          console.log(`      email: ${data.email || 'N/A'}`);
          console.log(`      role: ${data.role || 'N/A'}`);
          console.log(`      status: ${data.status || 'N/A'}`);
          console.log(`      Fields: ${Object.keys(data).join(', ')}`);
        });
      } else {
        console.log('   ⚠️  /users collection exists but is EMPTY');
      }
    } else {
      console.log('❌ /users COLLECTION DOES NOT EXIST\n');
      console.log('   The /users collection has not been created.');
      console.log('   All verified users need to be migrated.\n');
    }
    
    console.log('\n' + '─'.repeat(70));
    console.log('📊 STEP 3: Other collections summary');
    console.log('─'.repeat(70) + '\n');
    
    for (const collName of collectionNames) {
      if (collName !== 'users') {
        const collSnap = await getDocs(collection(db, collName));
        console.log(`  • ${collName}: ${collSnap.docs.length} document(s)`);
      }
    }
    
    console.log('\n' + '═'.repeat(70));
    console.log('AUDIT COMPLETE');
    console.log('═'.repeat(70) + '\n');
    
  } catch (error) {
    console.error('❌ AUDIT ERROR:', error.message);
  }
})();
